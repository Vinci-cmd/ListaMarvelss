      }

      return true;

      if (sortMode === 'rating') return (progress[b.id]?.rating || 0) - (progress[a.id]?.rating || 0);

                            </button>

          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-5">

                      </p>

                      {item.multiverseConnection && (
                        <div className="text-[10px] text-purple-300 bg-purple-500/10 border border-purple-500/25 rounded-lg px-2.5 py-1 mb-2.5 line-clamp-1 font-medium">
                          🔗 {item.multiverseConnection}
                        </div>
                      )}

                      {/* Personaggi principali */}
                      <div className="flex flex-wrap gap-1 mb-2">
                        {item.characters.slice(0, 3).map(char => (
                          <span
                            key={char}
                            className="text-[10px] bg-white/[0.04] border border-white/5 text-zinc-300 px-2 py-0.5 rounded-md truncate max-w-[130px]"
                          >
                            {char.split(' / ')[0]}
                          </span>
                        ))}
                        {item.characters.length > 3 && (
                          <span className="text-[10px] text-zinc-500 px-1 py-0.5">
                            +{item.characters.length - 3}
                          </span>
                        )}
                      </div>
                    </div>

                    {/* Footer Interattivo Card */}
                    <div className="pt-2.5 border-t border-white/10 space-y-2.5">
                      {/* Post-credit & Valutazione */}
                      <div className="flex items-center justify-between">
                        <span className="text-[11px] font-medium text-amber-400/90 flex items-center gap-1">
                          <Sparkles className="w-3 h-3" />
                          {item.postCreditsCount > 0 ? `${item.postCreditsCount} Post-Credit` : 'No Post-Credit'}
                        </span>

                        {/* Stelline Valutazione Rapida */}
                        <div className="flex items-center gap-0.5" onClick={(e) => e.stopPropagation()}>
                          {[1, 2, 3, 4, 5].map(star => (
                            <button
                              key={star}
                              type="button"
                              onClick={(e) => setItemRating(item.id, star === userItem.rating ? 0 : star, e)}
                              className="p-0.5 text-zinc-600 hover:text-amber-400 transition"
                            >
                              <Star
                                className={`w-3.5 h-3.5 ${
                                  star <= (userItem.rating || 0)
                                    ? 'fill-amber-400 text-amber-400'
                                    : ''
                                }`}
                              />
                            </button>
                          ))}
                        </div>
                      </div>

                      {/* Bottoni Azione: Segna Visto + Toggle Coppia */}
                      <div className="flex items-center gap-2">
                        <button
                          type="button"
                          onClick={(e) => toggleWatched(item.id, e)}
                          className={`flex-1 py-2 px-3 rounded-xl text-xs font-bold flex items-center justify-center gap-1.5 transition ${
                            isWatched
                              ? 'bg-emerald-500/20 text-emerald-300 border border-emerald-500/40 hover:bg-emerald-500/30'
                              : isInProgress
                              ? 'bg-amber-500/20 text-amber-300 border border-amber-500/40'
                              : 'bg-white/5 hover:bg-white/10 text-zinc-200 border border-white/10'
                          }`}
                        >
                          {isWatched ? (
                            <>
                              <CheckCircle2 className="w-4 h-4 text-emerald-400" />
                              <span>Visto</span>
                            </>
                          ) : isInProgress ? (
                            <>
                              <Clock className="w-4 h-4 text-amber-400" />
                              <span>In Corso</span>
                            </>
                          ) : (
                            <>
                              <Circle className="w-4 h-4 text-zinc-400" />
                              <span>Segna Visto</span>
                            </>
                          )}
                        </button>

                        {userProfile.coupleMode && (
                          <button
                            type="button"
                            onClick={(e) => togglePartnerWatched(item.id, e)}
                            title={
                              userItem.partnerWatched
                                ? `Visto anche da ${userProfile.partnerName}`
                                : `Non ancora visto da ${userProfile.partnerName}`
                            }
                            className={`p-2 rounded-xl border transition ${
                              userItem.partnerWatched
                                ? 'bg-rose-500/20 border-rose-500/40 text-rose-400'
                                : 'bg-white/5 border-white/10 text-zinc-500 hover:text-rose-300'
                            }`}
                          >
                            <Heart className={`w-4 h-4 ${userItem.partnerWatched ? 'fill-rose-500' : ''}`} />
                          </button>
                        )}
                      </div>
                    </div>
                  </div>
                </div>
              );
            })}
          </div>
        )}
      </main>

      {/* Modale Dettagli Completi Scheda Marvel */}
      {activeItem && (
        <div
          className="fixed inset-0 z-50 bg-black/80 backdrop-blur-md flex items-center justify-center p-4 overflow-y-auto"
          onClick={() => setActiveItem(null)}
        >
          <div
            className="bg-zinc-900 border border-white/15 rounded-3xl max-w-2xl w-full overflow-hidden shadow-2xl my-8"
            onClick={(e) => e.stopPropagation()}
          >
            {/* Header Modale */}
            <div className={`relative p-6 sm:p-8 bg-gradient-to-br ${activeItem.gradient} overflow-hidden`}>
              <PosterEmblem type={activeItem.iconType} className="w-48 h-48 opacity-20 absolute -right-6 -bottom-6 text-white" />

              <button
                onClick={() => setActiveItem(null)}
                className="absolute top-4 right-4 p-2 rounded-full bg-black/50 hover:bg-black/70 text-white border border-white/15 transition"
              >
                <X className="w-5 h-5" />
              </button>

              <div className="relative z-10">
                <div className="flex flex-wrap items-center gap-2 mb-2">
                  <span className="px-2.5 py-1 rounded-lg bg-black/60 text-xs font-bold text-white border border-white/15">
                    {activeItem.phase}
                  </span>
                  <span className="px-2.5 py-1 rounded-lg bg-black/60 text-xs font-semibold text-amber-300 border border-white/15">
                    {activeItem.saga}
                  </span>
                  <span className="px-2.5 py-1 rounded-lg bg-white/15 text-xs font-bold text-white">
                    {activeItem.type}
                  </span>
                </div>

                <h2 className="text-2xl sm:text-3xl font-black text-white leading-tight">
                  {activeItem.title}
                </h2>
                <p className="text-xs sm:text-sm text-white/80 mt-1 italic">
                  Titolo originale: {activeItem.originalTitle}
                </p>
              </div>
            </div>

            {/* Corpo Dettagli */}
            <div className="p-6 sm:p-8 space-y-6">
              {/* Griglia Metadati Chiave */}
              <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
                <div className="p-3 rounded-xl bg-black/40 border border-white/10">
                  <div className="text-[11px] text-zinc-400">Uscita Ufficiale</div>
                  <div className="text-sm font-bold text-white mt-0.5">{activeItem.releaseDate}</div>
                </div>
                <div className="p-3 rounded-xl bg-black/40 border border-white/10">
                  <div className="text-[11px] text-zinc-400">Anno Timeline</div>
                  <div className="text-sm font-bold text-amber-400 mt-0.5">{activeItem.timelineYear}</div>
                </div>
                <div className="p-3 rounded-xl bg-black/40 border border-white/10">
                  <div className="text-[11px] text-zinc-400">Durata / Episodi</div>
                  <div className="text-sm font-bold text-white mt-0.5">{activeItem.durationLabel}</div>
                </div>
                <div className="p-3 rounded-xl bg-black/40 border border-white/10">
                  <div className="text-[11px] text-zinc-400">Importanza</div>
                  <div className="text-sm font-bold text-emerald-400 mt-0.5">{activeItem.essentiality}</div>
                </div>
              </div>

              {/* Sinossi */}
              <div>
                <h4 className="text-xs font-bold uppercase tracking-wider text-zinc-400 mb-1.5">
                  Trama & Sinossi Ufficiale
                </h4>
                <p className="text-sm text-zinc-200 leading-relaxed">
                  {activeItem.synopsis}
                </p>
              </div>

              {/* Box Connessioni Multiverso (se presente) */}
              {activeItem.multiverseConnection && (
                <div className="p-3.5 rounded-xl bg-purple-500/10 border border-purple-500/30">
                  <div className="flex items-center gap-2 text-purple-300 font-bold text-xs uppercase tracking-wider mb-1">
                    <Layers className="w-4 h-4" />
                    <span>Connessioni Multiverso & Cameo MCU</span>
                  </div>
                  <p className="text-xs sm:text-sm text-purple-100/90 leading-relaxed">
                    {activeItem.multiverseConnection}
                  </p>
                </div>
              )}

              {/* Box Scene Post-Credit */}
              <div className="p-4 rounded-xl bg-amber-500/10 border border-amber-500/30">
                <div className="flex items-center gap-2 text-amber-300 font-bold text-xs uppercase tracking-wider mb-1">
                  <Sparkles className="w-4 h-4" />
                  <span>Guida Scene Post-Credit ({activeItem.postCreditsCount})</span>
                </div>
                <p className="text-xs sm:text-sm text-amber-100/90 leading-relaxed">
                  {activeItem.postCreditsHint}
                </p>
              </div>

              {/* Cast e Personaggi Principali */}
              <div>
                <h4 className="text-xs font-bold uppercase tracking-wider text-zinc-400 mb-2">
                  Protagonisti & Personaggi Chiave
                </h4>
                <div className="flex flex-wrap gap-2">
                  {activeItem.characters.map(char => (
                    <span
                      key={char}
                      className="px-3 py-1.5 rounded-xl bg-white/5 border border-white/10 text-xs font-medium text-zinc-200"
                    >
                      {char}
                    </span>
                  ))}
                </div>
              </div>

              {/* Sezione Tracciamento Utente nella Modale */}
              <div className="pt-4 border-t border-white/10 space-y-4">
                <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
                  <div>
                    <div className="text-xs font-bold uppercase tracking-wider text-zinc-400 mb-1.5">
                      Stato di Visione
                    </div>
                    <div className="flex items-center gap-2">
                      {[
                        { id: 'unwatched', label: 'Da Vedere' },
                        { id: 'in_progress', label: 'In Corso' },
                        { id: 'watched', label: 'Completato' }
                      ].map(st => {
                        const active = (progress[activeItem.id]?.status || 'unwatched') === st.id;
                        return (
                          <button
                            key={st.id}
                            type="button"
                            onClick={() => setItemStatus(activeItem.id, st.id)}
                            className={`px-3 py-1.5 rounded-xl text-xs font-bold border transition ${
                              active
                                ? 'bg-red-600 text-white border-red-500'
                                : 'bg-black/40 text-zinc-400 border-white/10 hover:text-white'
                            }`}
                          >
                            {st.label}
                          </button>
                        );
                      })}
                    </div>
                  </div>

                  {/* Voto */}
                  <div>
                    <div className="text-xs font-bold uppercase tracking-wider text-zinc-400 mb-1.5">
                      La tua Valutazione
                    </div>
                    <div className="flex items-center gap-1">
                      {[1, 2, 3, 4, 5].map(star => (
                        <button
                          key={star}
                          type="button"
                          onClick={(e) => setItemRating(activeItem.id, star, e)}
                          className="p-1 text-zinc-600 hover:text-amber-400 transition"
                        >
                          <Star
                            className={`w-6 h-6 ${
                              star <= (progress[activeItem.id]?.rating || 0)
                                ? 'fill-amber-400 text-amber-400'
                                : ''
                            }`}
                          />
                        </button>
                      ))}
                    </div>
                  </div>
                </div>

                {/* Note Personali */}
                <div>
                  <label className="block text-xs font-bold uppercase tracking-wider text-zinc-400 mb-1.5">
                    Note & Commenti Personali
                  </label>
                  <textarea
                    rows={2}
                    value={progress[activeItem.id]?.notes || ''}
                    onChange={(e) => updateItemNotes(activeItem.id, e.target.value)}
                    placeholder="Scrivi un appunto su questo film/serie, teorie sul multiverso o con chi l'hai visto..."
                    className="w-full px-3.5 py-2.5 rounded-xl bg-black/50 border border-white/10 text-sm text-white placeholder-zinc-500 focus:outline-none focus:border-red-500"
                  />
                </div>
              </div>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
